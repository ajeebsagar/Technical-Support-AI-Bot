# ─────────────────────────────────────────────────────────────────────────────
# Pinned, mutually-compatible versions for the Upwork API Support Bot.
# Tested on Python 3.11 (CPU only). On Streamlit Community Cloud, set the app's
# Python version to 3.11 (Advanced settings) so prebuilt wheels are used —
# newer versions (3.13/3.14) have no wheels for these pins and would try to
# compile from source and fail.
# ─────────────────────────────────────────────────────────────────────────────

# Install PyTorch from its CPU-only wheel index. Without this, `torch` pulls in
# ~2 GB of NVIDIA CUDA libraries that are useless on a CPU host and blow past
# Streamlit Cloud's disk/time limits. The "+cpu" local version is served ONLY
# from this index, so the resolver cannot accidentally grab the CUDA build.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.12.0+cpu

# PDF text extraction
pypdf==5.1.0

# LangChain core + the specific integration packages requested by the spec
langchain==0.3.13
langchain-text-splitters==0.3.4
langchain-chroma==0.1.4
langchain-huggingface==0.1.2

# Local embedding model (all-MiniLM-L6-v2) runs through sentence-transformers
sentence-transformers==3.3.1

# Vector database (persisted to ./chroma_db)
chromadb==0.5.23

# OpenAI-compatible client — pointed at DeepInfra's base_url
openai==1.58.1

# Loads DEEPINFRA_API_KEY from a local .env file (never committed)
python-dotenv==1.0.1

# Web UI
streamlit==1.41.1
